Example 3: Full component with modal
javascriptimport { useEffect, useState } from 'react';

const API_URL = 'https://6180u0u9xf.execute-api.us-east-1.amazonaws.com/prod';

function AdminDashboard() {
  const [users, setUsers] = useState([]);
  const [selectedUser, setSelectedUser] = useState(null);
  const [loading, setLoading] = useState(true);

  // Load all users
  useEffect(() => {
    async function loadUsers() {
      const response = await fetch(`${API_URL}/admin/users`);
      const data = await response.json();
      setUsers(data.users);
      setLoading(false);
    }
    loadUsers();
  }, []);

  // View single user
  async function viewUser(userId) {
    const response = await fetch(`${API_URL}/admin/users/${userId}`);
    const data = await response.json();
    setSelectedUser(data.user);
  }

  // Close modal
  function closeModal() {
    setSelectedUser(null);
  }

  if (loading) return <p>Loading users...</p>;

  return (
    <div>
      <h1>Admin Dashboard</h1>
      <p>Total Users: {users.length}</p>

      {/* Users Table */}
      <table>
        <thead>
          <tr>
            <th>Email</th>
            <th>Name</th>
            <th>Admin</th>
          </tr>
        </thead>
        <tbody>
          {users.map(user => (
            <tr key={user.user_id} onClick={() => viewUser(user.user_id)}>
              <td>{user.email}</td>
              <td>{user.first_name} {user.last_name}</td>
              <td>{user.is_admin ? 'Yes' : 'No'}</td>
            </tr>
          ))}
        </tbody>
      </table>

      {/* User Detail Modal */}
      {selectedUser && (
        <div className="modal">
          <h2>User Details</h2>
          <p><strong>Email:</strong> {selectedUser.email}</p>
          <p><strong>Name:</strong> {selectedUser.first_name} {selectedUser.last_name}</p>
          <p><strong>Favorite Notes:</strong> {selectedUser.favorite_notes || 'None'}</p>
          <p><strong>Admin:</strong> {selectedUser.is_admin ? 'Yes' : 'No'}</p>
          <p><strong>Created:</strong> {new Date(selectedUser.created_at).toLocaleDateString()}</p>
          <button onClick={closeModal}>Close</button>
        </div>
      )}
    </div>
  );
}

export default AdminDashboard;
